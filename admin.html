-- 1. Active RLS sur la table admins ALTER TABLE public.admins ENABLE ROW LEVEL SECURITY; -- 2. Supprime les anciennes politiques si elles existent DROP POLICY IF EXISTS "admins_read_all" ON public.admins; DROP POLICY IF EXISTS "admins_update_self" ON public.admins; DROP POLICY IF EXISTS "admins_insert_service" ON public.admins; DROP POLICY IF EXISTS "admins_delete_service" ON public.admins; -- 3. Politique : un utilisateur authentifié peut lire SA PROPRE ligne -- (et seulement la sienne) + les logistiques peuvent lire tout le monde. CREATE POLICY "admins_read_self_or_logistique" ON public.admins FOR SELECT TO authenticated USING ( email = auth.jwt() ->> 'email' OR EXISTS ( SELECT 1 FROM public.admins a WHERE a.email = auth.jwt() ->> 'email' AND a.role = 'logistique' ) ); -- 4. Politique : un utilisateur peut modifier uniquement sa propre ligne CREATE POLICY "admins_update_self" ON public.admins FOR UPDATE TO authenticated USING (email = auth.jwt() ->> 'email') WITH CHECK (email = auth.jwt() ->> 'email'); -- 5. Aucune insertion/suppression REST : passer par l'Edge Function sécurisée -- (déjà utilisée dans le code : /functions/v1/secure-admin-users) -- Donc on ne crée PAS de policy INSERT/DELETE pour authenticated. -- Elles resteront bloquées par défaut → comportement voulu. -- 6. Même logique pour missions_2 et demandes : RLS activée ALTER TABLE public.missions_2 ENABLE ROW LEVEL SECURITY; ALTER TABLE public.demandes ENABLE ROW LEVEL SECURITY; -- Lecture publique (nécessaire au formulaire public de demande) : à adapter -- Ici, on autorise la lecture à tous pour les listes, mais l'écriture passe -- par des rôles vérifiés (via l'Edge Function ou l'API anon pour les insertions -- publiques contrôlées par le formulaire). CREATE POLICY "missions_2_read_all" ON public.missions_2 FOR SELECT TO anon, authenticated USING (true); CREATE POLICY "demandes_read_all" ON public.demandes FOR SELECT TO anon, authenticated USING (true); -- ⚠️ Les PATCH et DELETE doivent passer par l'Edge Function ou par des -- policies basées sur le JWT. Pour éviter les suppressions anonymes, -- on bloque tout UPDATE/DELETE direct depuis anon : -- (Aucune policy UPDATE/DELETE pour anon → bloqué par défaut)
